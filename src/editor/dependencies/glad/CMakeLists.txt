
FetchContent_Declare(
	glad
	EXCLUDE_FROM_ALL
	GIT_REPOSITORY https://github.com/Dav1dde/glad.git
	GIT_TAG 73db193f853e2ee079bf3ca8a64aa2eaf6459043 # v2.0.8
	SOURCE_SUBDIR cmake
)
FetchContent_MakeAvailable(glad)

mark_as_advanced(FORCE
	FETCHCONTENT_UPDATES_DISCONNECTED_GLAD
	FETCHCONTENT_SOURCE_DIR_GLAD
	GLAD_SOURCES_DIR
) 
set(GLAD_SOURCES_DIR /home/bru/demoscene/external/glad-src/)
set(GLAD_AUTOGENERATED_DIR ${PROJECT_SOURCE_DIR}/external/glad-autogenerated)
add_subdirectory("${GLAD_SOURCES_DIR}/cmake" glad_cmake)
glad_add_library(glad STATIC EXCLUDE_FROM_ALL
	REPRODUCIBLE QUIET
	LOCATION ${GLAD_AUTOGENERATED_DIR}
	API gl:core=4.5)
target_include_directories(glad INTERFACE
	$<BUILD_INTERFACE:${GLAD_AUTOGENERATED_DIR}/include/glad>
	$<INSTALL_INTERFACE:glad>
)

add_library(glad_wrapper)
target_sources(glad_wrapper PUBLIC
	FILE_SET glad_wrapper TYPE CXX_MODULES
	FILES
	./glad.cpp
)

target_link_libraries(glad_wrapper
	PRIVATE glfw
	PRIVATE glfw_wrapper
	PRIVATE glad
)
